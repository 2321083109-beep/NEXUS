1. Instalar PHP

sudo apt update && sudo apt upgrade -y
sudo apt install php php-cli php-mysql php-curl php-mbstring php-xml php-zip -y

2. Levantar el servidor apache

sudo apt install apache2 libapache2-mod-php -y
sudo systemctl enable apache2
sudo systemctl start apache2
sudo systemctl status apache2

3. Ver la IP

ip a
hostname -I
curl ipconfig.me
curl ipinfo.io/ip
